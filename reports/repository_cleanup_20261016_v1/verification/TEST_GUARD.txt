import os, sys
from pathlib import Path
def offline_guard(event, args):
    if event == 'socket.connect':
        raise PermissionError('Repository cleanup tests cannot connect to a network')
    if event == 'subprocess.Popen':
        executable = os.fsdecode(args[0])
        if Path(executable).name in {'ssh','scp','sftp','rsync','tmux','curl','wget'}:
            raise PermissionError('Repository cleanup tests cannot launch remote or service tools')
sys.addaudithook(offline_guard)


import os,json,subprocess,hashlib

def rawrecord(p):
 try:
  b=Path(p).read_bytes();return {'path':str(p),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest(),'text':b.decode(errors='replace')}
 except OSError as e:return {'path':str(p),'error':type(e).__name__+': '+str(e)}
def command(args):
 try:
  r=subprocess.run(args,capture_output=True,timeout=25)
  return {'args':args,'actual_os_returncode':r.returncode,'stdout':r.stdout.decode(errors='replace'),'stderr':r.stderr.decode(errors='replace'),'stdout_sha256':hashlib.sha256(r.stdout).hexdigest(),'stderr_sha256':hashlib.sha256(r.stderr).hexdigest()}
 except (OSError,subprocess.TimeoutExpired) as e:return {'args':args,'error':type(e).__name__+': '+str(e)}

 ['date','--iso-8601=seconds'],['uptime','-s'],['last','-x','-n','12','reboot','shutdown'],['journalctl','--list-boots','--no-pager'],

 ['journalctl','--since','2026-10-19 07:00:00 UTC','--until','2026-10-19 11:00:00 UTC','--no-pager','-o','short-iso','--grep','(?i)(out of memory|oom-kill|killed process|systemd-oomd|reboot|shutdown|nvrm|xid|power key|sigkill|killall|pkill)'],
 [c['hosts']['5090']['tmux_binary'],'list-sessions']]]

for p in sorted(root.glob('*.log')):

def digest(data):return hashlib.sha256(data).hexdigest()
def identity(pid):
 try:
  stat=Path('/proc',str(pid),'stat').read_text();f=stat[stat.rfind(')')+2:].split()
  return {'pid':pid,'state':f[0],'ppid':int(f[1]),'pgid':int(f[2]),'start_ticks':int(f[19]),'argv':[a.decode() for a in Path('/proc',str(pid),'cmdline').read_bytes().split(b'\0') if a],'cwd':os.readlink('/proc/'+str(pid)+'/cwd')}
 except (FileNotFoundError,ProcessLookupError,PermissionError) as e:return {'pid':pid,'unavailable':type(e).__name__}

  return None
